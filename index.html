ALU_SOC_slave_controller_AHB

4-bit ALU with AHB-Lite single-transfer, zero-wait-state slave wrapper interface

Design

`timescale 1ns/1ps

module alu_4bit (
    input  wire [3:0] A,
    input  wire [3:0] B,
    input  wire [1:0] OP,
    output reg  [3:0] RESULT
);

    always @(*) begin
        case (OP)
            2'b00: RESULT = A + B;  
            2'b01: RESULT = A - B;  
            2'b10: RESULT = A & B;  
            2'b11: RESULT = A | B;  
            default: RESULT = 4'b0000;
        endcase
    end

endmodule

// AHB-LITE SLAVE WRAPPER FOR 4-BIT ALU with Single-transfer, zero-wait-state 


module ahb_alu_slave (

    input  wire        HCLK,
    input  wire        HRESETn,

    // AHB-Lite address/control phase
    input  wire        HSEL,
    input  wire [31:0] HADDR,
    input  wire [1:0]  HTRANS,
    input  wire        HWRITE,
    input  wire [2:0]  HSIZE,
    input  wire        HREADY,

    // AHB-Lite write-data phase
    input  wire [31:0] HWDATA,

    // AHB-Lite response/read-data signals
    output reg  [31:0] HRDATA,
    output wire        HREADYOUT,
    output wire        HRESP
);

    // Register address map

    localparam ADDR_A      = 32'h0000_0000;
    localparam ADDR_B      = 32'h0000_0004;
    localparam ADDR_OP     = 32'h0000_0008;
    localparam ADDR_RESULT = 32'h0000_000C;

    // Register storage

    reg [3:0] operand_A;
    reg [3:0] operand_B;
    reg [1:0] operation;

    wire [3:0] alu_result;

    // ALU instantiation

    alu_4bit U_ALU (
        .A      (operand_A),
        .B      (operand_B),
        .OP     (operation),
        .RESULT (alu_result)
    );

    // Address phase registers. These hold the transfer information for the following data phase.

    reg [31:0] addr_reg;
    reg        write_reg;
    reg        transfer_reg;

    wire valid_address;

    assign valid_address =
           (addr_reg == ADDR_A)      ||
           (addr_reg == ADDR_B)      ||
           (addr_reg == ADDR_OP)     ||
           (addr_reg == ADDR_RESULT);

    // Zero-wait-state response.
    // HRESP remains OKAY in this basic implementation.

    assign HREADYOUT = 1'b1;
    assign HRESP     = 1'b0;

    // Capture address/control during the address phase.
    // HTRANS[1] = 1 for NONSEQ and SEQ transfers.

    always @(posedge HCLK or negedge HRESETn) begin

        if (!HRESETn) begin
            addr_reg     <= 32'h0;
            write_reg    <= 1'b0;
            transfer_reg <= 1'b0;
        end
        else if (HREADY) begin

            transfer_reg <= HSEL && HTRANS[1];

            if (HSEL && HTRANS[1]) begin
                addr_reg  <= HADDR;
                write_reg <= HWRITE;
            end

        end
    end

    // Write registers during the data phase.
    // HWDATA corresponds to the previously captured address.

    always @(posedge HCLK or negedge HRESETn) begin

        if (!HRESETn) begin
            operand_A <= 4'b0000;
            operand_B <= 4'b0000;
            operation <= 2'b00;
        end
        else if (HREADY && transfer_reg &&
                 write_reg && valid_address) begin

            case (addr_reg)

                ADDR_A:
                    operand_A <= HWDATA[3:0];

                ADDR_B:
                    operand_B <= HWDATA[3:0];

                ADDR_OP:
                    operation <= HWDATA[1:0];

                ADDR_RESULT: begin
                    // RESULT is read-only.
                end

                default: begin
                    // No operation.
                end

            endcase
        end
    end

    // Read data phase.
    // HRDATA corresponds to the captured address.

    always @(*) begin

        HRDATA = 32'h0000_0000;

        if (transfer_reg && !write_reg && valid_address) begin

            case (addr_reg)

                ADDR_A:
                    HRDATA = {28'b0, operand_A};

                ADDR_B:
                    HRDATA = {28'b0, operand_B};

                ADDR_OP:
                    HRDATA = {30'b0, operation};

                ADDR_RESULT:
                    HRDATA = {28'b0, alu_result};

                default:
                    HRDATA = 32'h0000_0000;

            endcase
        end
    end

    // HSIZE is included for interface compatibility.

endmodule

module tb_ahb_alu_slave;

    // AHB-Lite signals

    reg         HCLK;
    reg         HRESETn;

    reg         HSEL;
    reg  [31:0] HADDR;
    reg  [1:0]  HTRANS;
    reg         HWRITE;
    reg  [2:0]  HSIZE;
    reg         HREADY;
    reg  [31:0] HWDATA;

    wire [31:0] HRDATA;
    wire        HREADYOUT;
    wire        HRESP;

    // Address map

    localparam ADDR_A      = 32'h0000_0000;
    localparam ADDR_B      = 32'h0000_0004;
    localparam ADDR_OP     = 32'h0000_0008;
    localparam ADDR_RESULT = 32'h0000_000C;

    // ALU operation codes

    localparam OP_ADD = 2'b00;
    localparam OP_SUB = 2'b01;
    localparam OP_AND = 2'b10;
    localparam OP_OR  = 2'b11;

    integer pass_count;
    integer fail_count;

    reg [31:0] read_value;

    // DUT instantiation

    ahb_alu_slave DUT (
        .HCLK      (HCLK),
        .HRESETn   (HRESETn),
        .HSEL      (HSEL),
        .HADDR     (HADDR),
        .HTRANS    (HTRANS),
        .HWRITE    (HWRITE),
        .HSIZE     (HSIZE),
        .HREADY    (HREADY),
        .HWDATA    (HWDATA),
        .HRDATA    (HRDATA),
        .HREADYOUT (HREADYOUT),
        .HRESP     (HRESP)
    );

    initial begin
        HCLK = 1'b0;
        forever #5 HCLK = ~HCLK;
    end

    initial begin
        $dumpfile("ahb_alu_waveform.vcd");
        $dumpvars(0, tb_ahb_alu_slave);
    end

    // Reset task

    task reset_dut;
    begin
        HRESETn = 1'b0;

        HSEL    = 1'b0;
        HADDR   = 32'h0;
        HTRANS  = 2'b00;
        HWRITE  = 1'b0;
        HSIZE   = 3'b010;
        HREADY  = 1'b1;
        HWDATA  = 32'h0;

        repeat (2) @(negedge HCLK);

        HRESETn = 1'b1;

        @(negedge HCLK);
      
        $display("AHB-Lite RESET COMPLETED");
     
    end
    endtask

    // AHB-Lite write task
    // Address phase followed by data phase

    task ahb_write;
        input [31:0] address;
        input [31:0] data;
    begin
        // ADDRESS PHASE

        @(negedge HCLK);

        HSEL   = 1'b1;
        HADDR  = address;
        HWRITE = 1'b1;
        HTRANS = 2'b10;       // NONSEQ
        HSIZE  = 3'b010;      // 32-bit word

        @(posedge HCLK);

        // DATA PHASE

        @(negedge HCLK);

        HWDATA = data;

        // No new address transfer

        HSEL   = 1'b0;
        HTRANS = 2'b00;
        HWRITE = 1'b0;
        HADDR  = 32'h0;

        @(posedge HCLK);

        if (HREADYOUT && !HRESP) begin
            $display("AHB WRITE: ADDR=%h DATA=%h",
                     address, data);
        end
        else begin
            $display("WRITE RESPONSE ERROR at %h", address);
            fail_count = fail_count + 1;
        end

        @(negedge HCLK);

        HWDATA = 32'h0;
    end
    endtask

    // AHB-Lite read task

    task ahb_read;
        input  [31:0] address;
        output [31:0] data;
    begin
        // ADDRESS PHASE

        @(negedge HCLK);

        HSEL   = 1'b1;
        HADDR  = address;
        HWRITE = 1'b0;
        HTRANS = 2'b10;       // NONSEQ
        HSIZE  = 3'b010;      // 32-bit word

        @(posedge HCLK);

        // DATA PHASE

        @(negedge HCLK);

        // HRDATA is valid for the captured address.
        // Allow combinational logic to settle.

        #1;
        data = HRDATA;

        if (HREADYOUT && !HRESP) begin
            $display("AHB READ : ADDR=%h DATA=%h",
                     address, data);
        end
        else begin
            $display("READ RESPONSE ERROR at %h", address);
            fail_count = fail_count + 1;
        end

        // End current transfer

        HSEL   = 1'b0;
        HTRANS = 2'b00;
        HADDR  = 32'h0;

        @(posedge HCLK);
        @(negedge HCLK);
    end
    endtask

    // ALU verification task

    task test_alu;
        input [3:0] A;
        input [3:0] B;
        input [1:0] OP;
        input [3:0] expected;

        reg [31:0] result;
    begin
        $display("ALU TEST: A=%d B=%d OP=%b", A, B, OP);

        // Write operands and operation

        ahb_write(ADDR_A, {28'b0, A});
        ahb_write(ADDR_B, {28'b0, B});
        ahb_write(ADDR_OP, {30'b0, OP});

        // Read ALU result

        ahb_read(ADDR_RESULT, result);

        // Check expected result

        if (result[3:0] === expected) begin
            $display("PASS: Expected=%h Received=%h",
                     expected, result[3:0]);
            pass_count = pass_count + 1;
        end
        else begin
            $display("FAIL: Expected=%h Received=%h",
                     expected, result[3:0]);
            fail_count = fail_count + 1;
        end
    end
    endtask

    // Main test sequence

    initial begin

        pass_count = 0;
        fail_count = 0;

        reset_dut;

        // TEST 1: ADD

        test_alu(4'd5, 4'd3, OP_ADD, 4'd8);

        // TEST 2: SUB
 
        test_alu(4'd9, 4'd4, OP_SUB, 4'd5);

        // TEST 3: AND

        test_alu(4'b1010, 4'b1100, OP_AND, 4'b1000);

        // TEST 4: OR

        test_alu(4'b1010, 4'b0101, OP_OR, 4'b1111);

        // TEST 5: REGISTER READ-BACK

        $display("");
        $display("REGISTER READ-BACK TEST");

        ahb_write(ADDR_A, 32'd7);
        ahb_read(ADDR_A, read_value);

        if (read_value[3:0] === 4'd7) begin
            $display("PASS: Operand A read-back");
            pass_count = pass_count + 1;
        end
        else begin
            $display("FAIL: Operand A read-back");
            fail_count = fail_count + 1;
        end

        // TEST 6: INVALID ADDRESS READ
        // The supplied RTL returns zero and HRESP=OKAY.

        $display("");
        $display("INVALID ADDRESS READ TEST");

        ahb_read(32'h0000_0020, read_value);

        if (read_value === 32'h0000_0000) begin
            $display("PASS: Invalid read returns zero");
            pass_count = pass_count + 1;
        end
        else begin
            $display("FAIL: Invalid read data");
            fail_count = fail_count + 1;
        end

        // TEST 7: INVALID ADDRESS WRITE
        // Invalid writes are ignored by the supplied RTL.

        $display("");
        $display("INVALID ADDRESS WRITE TEST");

        ahb_write(32'h0000_0020, 32'h0000_00AA);
        ahb_read(ADDR_A, read_value);

        if (read_value[3:0] === 4'd7) begin
            $display("PASS: Invalid write did not change A");
            pass_count = pass_count + 1;
        end
        else begin
            $display("FAIL: Invalid write affected A");
            fail_count = fail_count + 1;
        end

        $display("AHB-LITE ALU TEST SUMMARY");
        $display("PASSED = %0d", pass_count);
        $display("FAILED = %0d", fail_count);
      
        #10;
        $finish;
    end

endmodule

`timescale 1ns/1ps

// MODULE 1: SRAM MEMORY ARRAY
// 32 KB = 8192 words x 32 bits
// Synchronous read and byte-enabled synchronous write

module sram_memory_array (
    input              clk,
    input              mem_en,
    input              mem_we,
    input      [12:0]  mem_addr,
    input      [31:0]  mem_wdata,
    input      [3:0]   mem_wstrb,
    output reg [31:0]  mem_rdata
);

    reg [31:0] mem [0:8191];
    integer i;

    initial begin
        mem_rdata = 32'b0;
      for (i = 0; i < 8192; i = i + 1)
            mem[i] = 32'b0;
    end

    always @(posedge clk) begin
        if (mem_en) begin
            if (mem_we) begin
                // WSTRB[0] controls bits [7:0]
                // WSTRB[1] controls bits [15:8]
                // WSTRB[2] controls bits [23:16]
                // WSTRB[3] controls bits [31:24]

                if (mem_wstrb[0])
                    mem[mem_addr][7:0] <= mem_wdata[7:0];

                if (mem_wstrb[1])
                    mem[mem_addr][15:8] <= mem_wdata[15:8];

                if (mem_wstrb[2])
                    mem[mem_addr][23:16] <= mem_wdata[23:16];

                if (mem_wstrb[3])
                    mem[mem_addr][31:24] <= mem_wdata[31:24];
            end
            else begin
                // Synchronous SRAM read
                mem_rdata <= mem[mem_addr];
            end
        end
    end

endmodule

// MODULE 2: AXI4-LITE-STYLE SRAM CONTROLLER
// One outstanding transaction at a time.
// AW and W can arrive independently.

module sram_controller (
    input             ACLK,
    input             ARESETn,

    // AXI write address channel
    input      [31:0] S_AXI_AWADDR,
    input             S_AXI_AWVALID,
    output            S_AXI_AWREADY,

    // AXI write data channel
    input      [31:0] S_AXI_WDATA,
    input      [3:0]  S_AXI_WSTRB,
    input             S_AXI_WVALID,
    output            S_AXI_WREADY,

    // AXI write response channel
    output            S_AXI_BVALID,
    input             S_AXI_BREADY,
    output     [1:0]  S_AXI_BRESP,

    // AXI read address channel
    input      [31:0] S_AXI_ARADDR,
    input             S_AXI_ARVALID,
    output            S_AXI_ARREADY,

    // AXI read data channel
    output reg [31:0] S_AXI_RDATA,
    output            S_AXI_RVALID,
    input             S_AXI_RREADY,
    output     [1:0]  S_AXI_RRESP,

    // SRAM macro interface
    output            mem_en,
    output            mem_we,
    output     [12:0] mem_addr,
    output     [31:0] mem_wdata,
    output     [3:0]  mem_wstrb,
    input      [31:0] mem_rdata
);

    // FSM states
    localparam IDLE       = 3'd0;
    localparam WRITE_MEM  = 3'd1;
    localparam WRITE_RESP = 3'd2;
    localparam READ_MEM   = 3'd3;
    localparam READ_CAP   = 3'd4;
    localparam READ_RESP  = 3'd5;

    reg [2:0] state;

    // Hold independently accepted AW and W channels
    reg aw_pending;
    reg w_pending;

    reg [31:0] awaddr_reg;
    reg [31:0] wdata_reg;
    reg [3:0]  wstrb_reg;
    reg [31:0] araddr_reg;

    // AXI address is byte-addressed.
    // [14:2] selects one of 8192 32-bit SRAM words.
    wire [12:0] word_addr;
    assign word_addr = araddr_reg[14:2];

    // READY signals
    assign S_AXI_AWREADY =
        ARESETn && (state == IDLE) && !aw_pending;

    assign S_AXI_WREADY =
        ARESETn && (state == IDLE) && !w_pending;

    // Give an already pending/new write priority over a read.
    assign S_AXI_ARREADY =
        ARESETn && (state == IDLE) &&
        !aw_pending && !w_pending &&
        !S_AXI_AWVALID && !S_AXI_WVALID;

    // Responses
    assign S_AXI_BVALID = (state == WRITE_RESP);
    assign S_AXI_RVALID = (state == READ_RESP);

    // OKAY response
    assign S_AXI_BRESP = 2'b00;
    assign S_AXI_RRESP = 2'b00;

    // SRAM interface controls
    assign mem_en =
        (state == WRITE_MEM) || (state == READ_MEM);

    assign mem_we = (state == WRITE_MEM);

    assign mem_addr =
        (state == WRITE_MEM) ? awaddr_reg[14:2] :
        (state == READ_MEM)  ? word_addr :
                               13'b0;

    assign mem_wdata = wdata_reg;
    assign mem_wstrb = wstrb_reg;

    // Controller FSM
    always @(posedge ACLK or negedge ARESETn) begin
        if (!ARESETn) begin
            state      <= IDLE;
            aw_pending <= 1'b0;
            w_pending  <= 1'b0;

            awaddr_reg <= 32'b0;
            wdata_reg  <= 32'b0;
            wstrb_reg  <= 4'b0;
            araddr_reg <= 32'b0;
            S_AXI_RDATA <= 32'b0;
        end
        else begin
            case (state)

                IDLE: begin
                    // Capture write address independently
                    if (S_AXI_AWVALID && S_AXI_AWREADY) begin
                        awaddr_reg <= S_AXI_AWADDR;
                        aw_pending <= 1'b1;
                    end

                    // Capture write data independently
                    if (S_AXI_WVALID && S_AXI_WREADY) begin
                        wdata_reg <= S_AXI_WDATA;
                        wstrb_reg <= S_AXI_WSTRB;
                        w_pending <= 1'b1;
                    end

                    // Start write after both address and data
                    // have been accepted.
                    if ((aw_pending ||
                         (S_AXI_AWVALID && S_AXI_AWREADY)) &&
                        (w_pending ||
                         (S_AXI_WVALID && S_AXI_WREADY))) begin
                        state <= WRITE_MEM;
                    end
                    else if (S_AXI_ARVALID && S_AXI_ARREADY) begin
                        araddr_reg <= S_AXI_ARADDR;
                        state <= READ_MEM;
                    end
                end

                WRITE_MEM: begin
                    // SRAM writes at this rising edge.
                    // mem_en=1 and mem_we=1.
                    aw_pending <= 1'b0;
                    w_pending  <= 1'b0;
                    state <= WRITE_RESP;
                end

                WRITE_RESP: begin
                    // Keep BVALID asserted until accepted.
                    if (S_AXI_BREADY)
                        state <= IDLE;
                end

                READ_MEM: begin
                    // SRAM samples read address at this edge.
                    // Its registered read data becomes available
                    // after this edge.
                    state <= READ_CAP;
                end

                READ_CAP: begin
                    // Capture the data produced by SRAM.
                    S_AXI_RDATA <= mem_rdata;
                    state <= READ_RESP;
                end

                READ_RESP: begin
                    // Keep RVALID and RDATA stable until accepted.
                    if (S_AXI_RREADY)
                        state <= IDLE;
                end

                default: state <= IDLE;

            endcase
        end
    end

endmodule

module soc_sram_tb;

    reg ACLK;
    reg ARESETn;

    reg [31:0] S_AXI_AWADDR;
    reg        S_AXI_AWVALID;
    wire       S_AXI_AWREADY;

    reg [31:0] S_AXI_WDATA;
    reg [3:0]  S_AXI_WSTRB;
    reg        S_AXI_WVALID;
    wire       S_AXI_WREADY;

    wire       S_AXI_BVALID;
    reg        S_AXI_BREADY;
    wire [1:0] S_AXI_BRESP;

    reg [31:0] S_AXI_ARADDR;
    reg        S_AXI_ARVALID;
    wire       S_AXI_ARREADY;

    wire [31:0] S_AXI_RDATA;
    wire        S_AXI_RVALID;
    reg         S_AXI_RREADY;
    wire [1:0] S_AXI_RRESP;

    wire        mem_en;
    wire        mem_we;
    wire [12:0] mem_addr;
    wire [31:0] mem_wdata;
    wire [3:0]  mem_wstrb;
    wire [31:0] mem_rdata;

    reg [31:0] read_value;

    initial ACLK = 1'b0;
    always #5 ACLK = ~ACLK;

    sram_controller DUT_CONTROLLER (
        .ACLK(ACLK),
        .ARESETn(ARESETn),

        .S_AXI_AWADDR(S_AXI_AWADDR),
        .S_AXI_AWVALID(S_AXI_AWVALID),
        .S_AXI_AWREADY(S_AXI_AWREADY),

        .S_AXI_WDATA(S_AXI_WDATA),
        .S_AXI_WSTRB(S_AXI_WSTRB),
        .S_AXI_WVALID(S_AXI_WVALID),
        .S_AXI_WREADY(S_AXI_WREADY),

        .S_AXI_BVALID(S_AXI_BVALID),
        .S_AXI_BREADY(S_AXI_BREADY),
        .S_AXI_BRESP(S_AXI_BRESP),

        .S_AXI_ARADDR(S_AXI_ARADDR),
        .S_AXI_ARVALID(S_AXI_ARVALID),
        .S_AXI_ARREADY(S_AXI_ARREADY),

        .S_AXI_RDATA(S_AXI_RDATA),
        .S_AXI_RVALID(S_AXI_RVALID),
        .S_AXI_RREADY(S_AXI_RREADY),
        .S_AXI_RRESP(S_AXI_RRESP),

        .mem_en(mem_en),
        .mem_we(mem_we),
        .mem_addr(mem_addr),
        .mem_wdata(mem_wdata),
        .mem_wstrb(mem_wstrb),
        .mem_rdata(mem_rdata)
    );

    // Instantiate physical SRAM model
    sram_memory_array DUT_SRAM (
        .clk(ACLK),
        .mem_en(mem_en),
        .mem_we(mem_we),
        .mem_addr(mem_addr),
        .mem_wdata(mem_wdata),
        .mem_wstrb(mem_wstrb),
        .mem_rdata(mem_rdata)
    );

    task axi_write;
        input [31:0] address;
        input [31:0] data;
        input [3:0]  strobe;
        begin
            @(negedge ACLK);
            S_AXI_AWADDR  = address;
            S_AXI_AWVALID = 1'b1;
            S_AXI_WDATA   = data;
            S_AXI_WSTRB   = strobe;
            S_AXI_WVALID  = 1'b1;

            // Wait until both channels are accepted.
            begin : WAIT_WRITE_ACCEPT
                reg aw_done;
                reg w_done;
                aw_done = 1'b0;
                w_done  = 1'b0;

                while (!(aw_done && w_done)) begin
                    @(posedge ACLK);
                    if (S_AXI_AWVALID && S_AXI_AWREADY)
                        aw_done = 1'b1;
                    if (S_AXI_WVALID && S_AXI_WREADY)
                        w_done = 1'b1;
                    @(negedge ACLK);
                    if (aw_done)
                        S_AXI_AWVALID = 1'b0;
                    if (w_done)
                        S_AXI_WVALID = 1'b0;
                end
            end

            // Wait for write response
            while (!S_AXI_BVALID)
                @(negedge ACLK);

            if (S_AXI_BRESP != 2'b00)
                $display("ERROR: Write response error!");

            S_AXI_BREADY = 1'b1;
            @(posedge ACLK);
            @(negedge ACLK);
            S_AXI_BREADY = 1'b0;

            $display(
                "[%0t] WRITE: ADDR=%h DATA=%h WSTRB=%b",
                $time, address, data, strobe
            );
        end
    endtask

    // AXI READ TASK
    task axi_read;
        input  [31:0] address;
        output [31:0] data;
        begin
            @(negedge ACLK);
            S_AXI_ARADDR  = address;
            S_AXI_ARVALID = 1'b1;

            // Address handshake
            begin : WAIT_READ_ADDRESS
                reg ar_done;
                ar_done = 1'b0;
                while (!ar_done) begin
                    @(posedge ACLK);
                    if (S_AXI_ARVALID && S_AXI_ARREADY)

module alu_core (
    input  wire [31:0] a,
    input  wire [31:0] b,
    input  wire [1:0]  op,
    output reg  [31:0] result
);
    always @(*) begin
        case (op)
            2'b00: result = a + b;
            2'b01: result = a - b;
            2'b10: result = a & b;
            2'b11: result = a | b;
            default: result = 32'b0;
        endcase
    end

module axi_alu_top (
    input  wire        aclk,
    input  wire        aresetn,

    // Write Address Channel
    input  wire [31:0] awaddr,
    input  wire        awvalid,
    output wire        awready,

    // Write Data Channel
    input  wire [31:0] wdata,
    input  wire        wvalid,
    output wire        wready,

    // Write Response Channel
    output reg  [1:0]  bresp,
    output reg         bvalid,
    input  wire        bready,

    // Read Address Channel
    input  wire [31:0] araddr,
    input  wire        arvalid,
    output wire        arready,

    // Read Data Channel
    output reg  [31:0] rdata,
    output reg  [1:0]  rresp,
    output reg         rvalid,
    input  wire        rready
);

    reg [31:0] reg_a, reg_b;
    reg [1:0]  reg_op;
    wire [31:0] alu_result;

    // Independently buffered write channels
    reg [31:0] saved_awaddr, saved_wdata;
    reg aw_pending, w_pending;

    assign awready = !aw_pending && !bvalid;
    assign wready  = !w_pending  && !bvalid;
    assign arready = !rvalid;

    alu_core u_alu (
        .a      (reg_a),
        .b      (reg_b),
        .op     (reg_op),
        .result (alu_result)
    );

    // AXI-like write interface
    always @(posedge aclk or negedge aresetn) begin
        if (!aresetn) begin
            reg_a       <= 32'b0;
            reg_b       <= 32'b0;
            reg_op      <= 2'b0;
            saved_awaddr <= 32'b0;
            saved_wdata  <= 32'b0;
            aw_pending  <= 1'b0;
            w_pending   <= 1'b0;
            bvalid      <= 1'b0;
            bresp       <= 2'b00;
        end else begin

            // Capture address independently
            if (awvalid && awready) begin
                saved_awaddr <= awaddr;
                aw_pending   <= 1'b1;
            end

            // Capture write data independently
            if (wvalid && wready) begin
                saved_wdata <= wdata;
                w_pending   <= 1'b1;
            end

            // Execute write once both channels have been captured.
            // Nonblocking assignments mean this uses previously
            // captured address/data on the following clock edge.
            if (aw_pending && w_pending && !bvalid) begin
                case (saved_awaddr[3:0])
                    4'h0: reg_a  <= saved_wdata;
                    4'h4: reg_b  <= saved_wdata;
                    4'h8: reg_op <= saved_wdata[1:0];
                    default: ;
                endcase

                aw_pending <= 1'b0;
                w_pending  <= 1'b0;
                bvalid     <= 1'b1;
                bresp      <= (saved_awaddr[3:0] == 4'h0 ||
                               saved_awaddr[3:0] == 4'h4 ||
                               saved_awaddr[3:0] == 4'h8)
                              ? 2'b00 : 2'b10;
            end

            // Hold response until accepted
            if (bvalid && bready)
                bvalid <= 1'b0;
        end
    end

    // Read interface
    always @(posedge aclk or negedge aresetn) begin
        if (!aresetn) begin
            rdata  <= 32'b0;
            rresp  <= 2'b00;
            rvalid <= 1'b0;
        end else begin
            // Hold read data stable while RVALID=1 and RREADY=0
            if (arvalid && arready) begin
                case (araddr[3:0])
                    4'h0: rdata <= reg_a;
                    4'h4: rdata <= reg_b;
                    4'h8: rdata <= {30'b0, reg_op};
                    4'hC: rdata <= alu_result;
                    default: rdata <= 32'hDEAD_BEEF;
                endcase

                rresp  <= (araddr[3:0] == 4'h0 ||
                           araddr[3:0] == 4'h4 ||
                           araddr[3:0] == 4'h8 ||
                           araddr[3:0] == 4'hC)
                          ? 2'b00 : 2'b10;
                rvalid <= 1'b1;
            end else if (rvalid && rready) begin
                rvalid <= 1'b0;
            end
        end
    end

endmodule

            2'b01: RESULT = A - B;  // SUB

module apb_alu_slave (

    // APB interface
    input  wire        PCLK,
    input  wire        PRESETn,

    input  wire [31:0] PADDR,
    input  wire        PWRITE,
    input  wire        PSEL,
    input  wire        PENABLE,
    input  wire [31:0] PWDATA,

    output wire [31:0] PRDATA,
    output wire        PREADY,
    output wire        PSLVERR
);

    // Register storage

    reg [3:0] operand_A;
    reg [3:0] operand_B;
    reg [1:0] operation;

    wire [3:0] alu_result;

    // Address definitions

    localparam ADDR_A      = 32'h0000_0000;
    localparam ADDR_B      = 32'h0000_0004;
    localparam ADDR_OP     = 32'h0000_0008;
    localparam ADDR_RESULT = 32'h0000_000C;

    // ALU IP instantiation

    alu_4bit U_ALU (

        .A      (operand_A),
        .B      (operand_B),
        .OP     (operation),
        .RESULT (alu_result)

    );

    // APB READY
    // Zero-wait-state slave

    assign PREADY = PSEL && PENABLE;

    // Address validity

    wire valid_address;

    assign valid_address =
           (PADDR == ADDR_A)      ||
           (PADDR == ADDR_B)      ||
           (PADDR == ADDR_OP)     ||
           (PADDR == ADDR_RESULT);

    // APB ERROR
  
    assign PSLVERR =
           PSEL && PENABLE && !valid_address;

    // WRITE OPERATION
   
    always @(posedge PCLK) begin

        if (!PRESETn) begin

            operand_A <= 4'b0000;
            operand_B <= 4'b0000;
            operation <= 2'b00;

        end

        else begin

            if (PSEL && PENABLE && PWRITE) begin

                case (PADDR)

                    ADDR_A: begin
                        operand_A <= PWDATA[3:0];
                    end

                    ADDR_B: begin
                        operand_B <= PWDATA[3:0];
                    end

                    ADDR_OP: begin
                        operation <= PWDATA[1:0];
                    end

                    ADDR_RESULT: begin
                        // RESULT is read-only
                    end

                    default: begin
                        // Invalid address
                    end

                endcase

            end

        end

    end

    // READ OPERATION
   
    reg [31:0] read_data;

    always @(*) begin

        read_data = 32'h0000_0000;

        if (PSEL && PENABLE && !PWRITE) begin

                    read_data = {28'b0, operand_B};

                    read_data = {28'b0, alu_result};
                end

                default: begin
                    read_data = 32'h0000_0000;
                end

            endcase

        end

    end

    assign PRDATA = read_data;

              FETCH    = 4'b0001,
              DECODE   = 4'b0010,
              INP      = 4'b0011,
              OUT      = 4'b0100,
              INCR     = 4'b0101,
              CLR      = 4'b0110,
              JZ_CHECK = 4'b0111,
              JZ_LOAD  = 4'b1000,
              HALT     = 4'b1001;

    // 1. PRESENT STATE REGISTER    
    // Sequential logic
    // Sensitivity: clock and asynchronous reset
   

    always @(posedge clk or posedge rst)
    begin
        if (rst)
            present_state <= START;
        else
            present_state <= next_state;
    end

    // 2. NEXT STATE LOGIC
    // Combinational logic
    // @(*) automatically includes all signals read inside this block: present_state, IR and Z.
    
    always @(*)
    begin

        // Default next state
        next_state = START;

        case (present_state)

            START:
            begin
                next_state = FETCH;
            end

            FETCH:
            begin
                next_state = DECODE;
            end

            DECODE:
            begin

                case (IR)

                    3'b000:                 // INP A
                        next_state = INP;

                    3'b001:                 // OUT A
                        next_state = OUT;

                    3'b010:                 // INCR A
                        next_state = INCR;

                    3'b011:                 // CLR A
                        next_state = CLR;

                    3'b100:                 // JZ aaaa
                        next_state = JZ_CHECK;

                    3'b101:                 // HALT
                        next_state = HALT;

                    default:                // Unused opcode
                        next_state = FETCH;

                endcase

            end

            INP:
            begin
                next_state = FETCH;
            end

            OUT:
            begin
                next_state = FETCH;
            end

            INCR:
            begin
                next_state = FETCH;
            end

            CLR:
            begin
                next_state = FETCH;
            end

            // JZ CHECK
            // Z affects ONLY the next state.
            // No output depends directly on Z.
          
           JZ_CHECK:
            begin

                if (Z == 1'b1)
                    next_state = JZ_LOAD;
                else
                    next_state = FETCH;

            end

            // JZ LOAD
            // PCload will be generated because the FSM is in JZ_LOAD state.
        
            JZ_LOAD:
            begin
                next_state = FETCH;
            end

            HALT:
            begin
                // Remain permanently in HALT
                next_state = HALT;
            end

            default:
            begin
                next_state = START;
            end

        endcase

    end

    // 3. OUTPUT LOGIC: Pure Moore logic: OUTPUT = f(PRESENT_STATE)
    // Z and IR are NOT used in this block.

    always @(*)
    begin

      // DEFAULT OUTPUT VALUES
     
        IRload     = 1'b0;
        PCload     = 1'b0;
        INmux      = 1'b0;
        OUTmux     = 1'b0;
        ALUcontrol = 2'b00;
        JZ         = 1'b0;
        Halt       = 1'b0;


        case (present_state)

            START:
            begin
                 // No control signal
            end

            FETCH:
            begin
                IRload = 1'b1;
            end

            DECODE:
            begin
                // No control signal
            end

            INP:
            begin
                INmux = 1'b1;
            end

            OUT:
            begin
                OUTmux = 1'b1;
            end

            INCR:
            begin
                ALUcontrol = 2'b01;
            end

            CLR:
            begin
                ALUcontrol = 2'b10;
            end

            JZ_CHECK:
            begin
                JZ = 1'b1;
            end

          JZ_LOAD:
            begin
                JZ     = 1'b1;
                PCload = 1'b1;
            end

            HALT:
            begin
                Halt = 1'b1;
            end

            default:
            begin
                IRload     = 1'b0;
                PCload     = 1'b0;
                INmux      = 1'b0;
                OUTmux     = 1'b0;
                ALUcontrol = 2'b00;
                JZ         = 1'b0;

// APB SLAVE WRAPPER (The Standard Interface)

module apb_soc_adder(
  
    // Core APB System Interface Ports
    input  wire        PCLK,       // Bus Clock
    input  wire        PRESETn,    // Active-LOW Bus Reset
    
    // APB Master Control Command Bus
    input  wire [31:0] PADDR,      // 32-bit Address Bus
    input  wire        PWRITE,     // 1 = Write, 0 = Read
    input  wire        PSEL,       // Peripheral Selection Line
    input  wire        PENABLE,    // Strobe / Access Phase Enable
    input  wire [31:0] PWDATA,     // Write Data Payload Bus
    
    // APB Response Output Bus back to Master
    output reg  [31:0] PRDATA,     // Read Data Return Bus
    output wire        PREADY,     // Slave Ready Status Handshake
    output reg         PSLVERR     // Slave Protocol Error Flag
);
  
    // Internal Wrapper Peripheral Storage Registers
    reg [31:0] reg1; // Offset 0x00: Operand A
    reg [31:0] reg2; // Offset 0x04: Operand B
    reg [31:0] reg4; // Offset 0x0C: Scratchpad Register
    
    // Interconnect wire between Core IP output and Wrapper Mux
    wire [8:0] w_sum;
  
    // Standardized Memory Map Registers Offset Addressing
    parameter ADDR_REG1 = 4'h0,  // Operand A
              ADDR_REG2 = 4'h4,  // Operand B
              ADDR_REG3 = 4'h8,  // Read-Only Result Zone
              ADDR_REG4 = 4'hC;  // Scratchpad
  
    // APB3 Handshake Engine: This slave is always ready in 2 cycles (zero wait-states)
    assign PREADY = 1'b1; 
  
    // APB REGISTER WRITE INTERFACE LOGIC 
    always @(posedge PCLK or negedge PRESETn) begin
        if (!PRESETn) begin
            reg1    <= 32'h0;
            reg2    <= 32'h0;
            reg4    <= 32'h0;
            PSLVERR <= 1'b0;
        end  
        else begin
            PSLVERR <= 1'b0; // Default clear error flag

            // APB Write must strictly occur only when PSEL=1, PENABLE=1, and PWRITE=1
            if (PSEL && PWRITE && PENABLE) begin
                case (PADDR[3:0]) // Decoding the lowest nibble for offset mapping
                    ADDR_REG1: reg1 <= PWDATA;
                    ADDR_REG2: reg2 <= PWDATA;
                    ADDR_REG4: reg4 <= PWDATA;
                    ADDR_REG3: PSLVERR <= 1'b1; // Error: Attempted to write to a read-only result zone
                    default:   PSLVERR <= 1'b1; // Error: Address out-of-bounds
                endcase
            end
        end
    end

    // APB REGISTER READ INTERFACE LOGIC 
    always @(posedge PCLK or negedge PRESETn) begin
        if (!PRESETn) begin
            PRDATA <= 32'h0;
        end  
        else begin
            // APB Read occurs during the active selection window when PWRITE=0
            if (PSEL && !PWRITE) begin
                case (PADDR[3:0])
                    ADDR_REG1: PRDATA <= reg1;
                    ADDR_REG2: PRDATA <= reg2;
                    // Zero-extend the 9-bit core sum to match 32-bit data lines
                    ADDR_REG3: PRDATA <= {23'b0, w_sum}; 
                    ADDR_REG4: PRDATA <= reg4;
                    default: begin
                        PRDATA  <= 32'hDEADBEEF; // Error feedback signature
                        // Note: PSLVERR will be set high in the synchronous block above
                    end
                endcase
            end
            else begin
                PRDATA <= 32'h0; // Maintain zero baseline if not targeted
            end
        end
    end
  
    // CORE IP HARDWARE INSTANTIATION
    my_adder1 inst1 (
        .a(reg1[7:0]),
        .b(reg2[7:0]),
        .sum(w_sum)
    );
  
endmodule

    input clk, rst, sel, wen,
    input [19:0] addr,
    input [31:0] wr_data,
    output reg [31:0] rd_data
);
  
    reg [31:0] reg1;
    reg [31:0] reg2;
    reg [31:0] reg3;
    reg [31:0] reg4;
  
    wire [8:0] w_sum;
  
    parameter ADDR_REG1 = 20'd0,
              ADDR_REG2 = 20'd4,
              ADDR_REG3 = 20'd8,
              ADDR_REG4 = 20'd12;
  
    always @(posedge clk) begin
        if (rst) begin
            reg1 <= 0;
            reg2 <= 0;
            reg3 <= 0;
            reg4 <= 0;
        end  
        else begin
            if (sel & wen) begin
                case (addr)
                    ADDR_REG1: reg1 <= wr_data;
                    ADDR_REG2: reg2 <= wr_data;
                    ADDR_REG3: reg3 <= wr_data;
                    ADDR_REG4: reg4 <= wr_data;
                    default:   reg1 <= wr_data;
                endcase
            end
        end
    end
 
    always @(posedge clk) begin
        if (rst) begin
            rd_data <= 0;
        end  
        else begin
            if (sel & (~wen)) begin
                case (addr)
                    ADDR_REG1: rd_data <= reg1;
                    ADDR_REG2: rd_data <= reg2;
                    ADDR_REG3: rd_data <= w_sum;
                    ADDR_REG4: rd_data <= reg4;
                    default:   rd_data <= reg1;
                endcase
            end
        end
    end
  
    my_adder1 inst1 (
        .a(reg1[7:0]),
